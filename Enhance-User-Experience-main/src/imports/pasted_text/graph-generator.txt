from flask import Flask, render_template, jsonify
import pandas as pd
import random

app = Flask(__name__)

relations = ["belongs_to", "depends_on", "connected_to", "linked_to"]

data = []

for i in range(1, 501):
    child = f"Node_{i}"
    parent = f"Node_{random.randint(0, i-1)}"
    relation = random.choice(relations)
    data.append([child, parent, relation])

df = pd.DataFrame(data, columns=["child", "parent", "relation"])

all_children = set(df["child"])
all_parents = set(df["parent"])
root_nodes = list(all_parents - all_children)

@app.route("/")
def index():
    return render_template("index.html")

@app.route("/graph")
def graph():
    return jsonify({
        "nodes": [{"id": n, "label": n} for n in root_nodes[:10]],
        "edges": []
    })

@app.route("/expand/<node_id>")
def expand(node_id):
    edges = []

    for _, row in df[df["parent"] == node_id].iterrows():
        edges.append({"from": row["parent"], "to": row["child"], "label": row["relation"]})

    for _, row in df[df["child"] == node_id].iterrows():
        edges.append({"from": row["parent"], "to": row["child"], "label": row["relation"]})

    return jsonify({"edges": edges})

@app.route("/full_lineage/<node_id>")
def full_lineage(node_id):

    lineage_nodes = set()
    lineage_edges = []

    current = node_id

    while True:
        parent = df[df["child"] == current]
        if parent.empty:
            break

        row = parent.iloc[0]

        lineage_nodes.add(row["parent"])
        lineage_nodes.add(row["child"])

        lineage_edges.append({
            "from": row["parent"],
            "to": row["child"],
            "label": row["relation"]
        })

        current = row["parent"]

    for _, row in df[df["parent"] == node_id].iterrows():
        lineage_nodes.add(row["parent"])
        lineage_nodes.add(row["child"])

        lineage_edges.append({
            "from": row["parent"],
            "to": row["child"],
            "label": row["relation"]
        })

    return jsonify({
        "nodes": list(lineage_nodes),
        "edges": lineage_edges
    })

@app.route("/path/<node_id>")
def path(node_id):
    path = []
    current = node_id

    while True:
        parent = df[df["child"] == current]
        if parent.empty:
            break

        p = parent.iloc[0]["parent"]
        path.append(p)
        current = p

    return jsonify(path[::-1])

@app.route("/all_nodes")
def all_nodes():
    return jsonify(list(set(df["child"]).union(set(df["parent"]))))

if __name__ == "__main__":
    app.run(debug=True)
